  <action>
    Replace the current app.vue (minimal plain nav) with a cyberpunk-styled global shell:

    ```vue
    <script setup lang="ts">
    function handleError(error: Error) {
      console.error('[KlubHub] Unhandled error:', error)
    }
    </script>

    <template>
      <div class="bg-surface min-h-screen font-data text-on-surface">
        <!-- Global navigation -->
        <nav class="bg-surface-container-high border-b border-outline-variant/20 px-6 py-3 flex items-center gap-6">
          <span class="font-command tracking-command text-primary text-lg font-semibold">
            KLUBHUB<span class="text-tertiary">.</span>DJ
          </span>
          <div class="flex gap-4 ml-8">
            <NuxtLink
              to="/tracklist"
              class="font-terminal tracking-terminal text-xs uppercase text-on-surface-variant hover:text-primary transition-colors duration-200"
              active-class="text-primary luminous-threshold px-2 py-1"
            >
              TRACKLIST
            </NuxtLink>
          </div>
        </nav>

        <!-- Page content with error boundary -->
        <NuxtErrorBoundary @error="handleError">
          <NuxtPage />
          <template #error="{ error, clearError }">
            <div class="flex items-center justify-center min-h-[60vh] p-8">
              <div class="glass-panel-heavy luminous-threshold p-8 max-w-lg w-full space-y-4">
                <h1 class="font-command tracking-command text-error text-2xl font-bold uppercase">
                  SYSTEM ERROR
                </h1>
                <p class="font-terminal tracking-terminal text-on-surface-variant text-xs uppercase leading-relaxed">
                  {{ error.message || 'An unexpected error occurred. Please reinitialize.' }}
                </p>
                <button
                  class="gradient-cta text-on-primary px-6 py-3 font-terminal tracking-terminal text-xs uppercase font-semibold w-full hover:shadow-glow-primary-strong transition-shadow duration-200"
                  @click="clearError()"
                >
                  REINITIALIZE
                </button>
              </div>
            </div>
          </template>
        </NuxtErrorBoundary>

        <!-- Toast notification system — ClientOnly prevents SSR hydration mismatch -->
        <ClientOnly>
          <Toaster />
        </ClientOnly>
      </div>
    </template>
    ```

    - NuxtLink `active-class` adds luminous-threshold visual to the active route
    - Future phases will add more nav links to the same nav element
    - Remove the `/about` link from the original nav (that page is not meaningful)
