<objective>
Build RA (Resident Advisor) integration for KlubHub DJ — enable importing artist data from RA into EPK and importing RA events as gigs.

Purpose: RA is the primary platform for electronic music artists. Integrating RA data enriches the press kit with verified bio, social links, and event history. Gig tracker can import RA events as gig records.

Output: RA Go client, EPK import UI, Gig import UI, REST endpoints
</objective>

<context>
# RA GraphQL API (unofficial, no auth)
- Endpoint: POST https://ra.co/graphql
- Content-Type: application/json
- Body: { "query": "...", "variables": { ... } }
- No authentication required for public queries

# Available queries (from research):
- artist(slug: String!): Artist — get artist by URL slug
- artist(id: ID!, slug: String): Artist — get artist by ID
- eventListings(filters: FilterInputDtoInput, pageSize: Int, page: Int): EventListingResponse — list events
- event(id: ID!): Event — get single event
- venue(id: ID!): Venue — get venue details
- areas(searchTerm: String): [Area] — search areas/cities

# Artist object fields (partial):
- id, name, slug, url
- biography { blurb, full }
- aliases
- social links: facebook, twitter, instagram, soundcloud, bandcamp, discogs, website
- artistAreas: [{ areaId, areaName, countryId, countryUrl }]
- artistVenues: [{ venueId, venueName }]
- events: [{ eventId, eventName, date, venueId, venueName, artists }]

# Event object fields (partial):
- id, title, date, startTime, endTime
- venue { name, id, address }
- artists: [{ name, id, url }]
- attending (guest count)
- contentUrl (ticket link)

# Known limitations:
- No official API documentation — GraphQL schema discovered via browser dev tools
- Implicit rate limits — respect with caching and reasonable request spacing
- Artist slug is the URL-safe name from ra.co/dj/{slug}
</context>

```go
type RAArtist struct {
    ID          string    `json:"id"`
    Name        string    `json:"name"`
    Slug        string    `json:"slug"`
    URL         string    `json:"url"`
    Biography   string    `json:"biography"` // blurb text
    Aliases     []string  `json:"aliases"`
    Facebook    string    `json:"facebook"`
    Twitter     string    `json:"twitter"`
    Instagram   string    `json:"instagram"`
    Soundcloud  string    `json:"soundcloud"`
    Bandcamp    string    `json:"bandcamp"`
    Discogs     string    `json:"discogs"`
    Website     string    `json:"website"`
    Areas       []RAArea  `json:"artistAreas"`
    Venues      []RAVenue `json:"artistVenues"`
    Followers   int       `json:"followerCount"`
}

type RAEVENT struct {
    ID          string    `json:"id"`
    Title       string    `json:"title"`
    Date        string    `json:"date"` // ISO format
    StartTime   string    `json:"startTime"`
    EndTime     string    `json:"endTime"`
    VenueID     string    `json:"venueId"`
    VenueName   string    `json:"venueName"`
    Artists     []RAArtistRef `json:"artists"`
    Attending   int       `json:"attending"`
    ContentURL  string    `json:"contentUrl"`
}

type RAArea struct {
    AreaID    string `json:"areaId"`
    AreaName  string `json:"areaName"`
    CountryID string `json:"countryId"`
    Country   string `json:"countryUrl"`
}

type RAVenue struct {
    VenueID   string `json:"venueId"`
    VenueName string `json:"venueName"`
}

type RAArtistRef struct {
    ArtistID string `json:"artistId"`
    Name     string `json:"name"`
    URL      string `json:"url"`
}
```

```go
type RAClient struct {

        cache:      NewRACache(),
    }
}

// GetArtist fetches artist by slug
func (c *RAClient) GetArtist(ctx context.Context, slug string) (*RAArtist, error)
```

Create apps/dj/app/stores/ra.ts:

```typescript
export const useRAStore = defineStore('ra', () => {
  const artist = ref<RAArtist | null>(null)
  const loading = ref(false)
  const error = ref<string | null>(null)

  async function fetchArtist(slug: string): Promise<boolean> {
    loading.value = true
    error.value = null
    try {
      const result = await $fetch<{ data: RAArtist }>(`/api/v1/epk/import-ra`, {
        method: 'POST',
        body: { artist_slug: slug }
      })
      artist.value = result.data
      return true
    } catch (e) {
      error.value = e.message
      return false
    } finally {
      loading.value = false
    }
  }

  function clear() {
    artist.value = null
    error.value = null
  }

  return { artist, loading, error, fetchArtist, clear }
})
```
