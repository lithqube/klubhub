<!-- API contracts Plan 02 calls. These are defined by Plan 01's handler. -->

POST /api/v1/tracklists/upload
  - multipart/form-data, field "file"
  - Response 201: { tracklist: Tracklist, tracks: Track[], warnings: ParseWarning[] }
  - Response 413: { error: "file_too_large", message: string }
  - Response 422: { error: "invalid_format"|"zero_tracks", message: string }

GET /api/v1/tracklists
  - Response 200: { tracklists: Tracklist[] }

GET /api/v1/tracklists/{id}
  - Response 200: { tracklist: Tracklist, tracks: Track[] }

PUT /api/v1/tracklists/{id}/tracks/{track_id}
  - Body: { title?: string, artist?: string, bpm?: number, musical_key?: string, album?: string, genre?: string }
  - Response 200: { track: Track }

PUT /api/v1/tracklists/{id}/tracks/{track_id}/artwork  (TRKL-14)
  - multipart/form-data, field "file" (image/jpeg, image/png, image/webp; max 10 MB)
  - Response 200: { artwork_url: string, artwork_source: "manual", artwork_status: "manual" }
  - Response 404: tracklist or track not found

DELETE /api/v1/tracklists/{id}
  - Response 204

GET /api/v1/settings
  - Response 200: UserSettings (djName, logoPath, defaultColors, tracklist_preferences JSONB, custom_placeholder_path TEXT)

PUT /api/v1/settings
  - Body: { tracklist_preferences: { preset, bg_mode, visible_fields, max_tracks }, custom_placeholder_path?: string }

```typescript
interface Tracklist {
  id: string
  title: string
  sourceFormat: string
  rawFilePath: string | null
  preset: string
  visibleFields: string[]
  bgMode: 'solid' | 'upload' | 'mosaic'
  bgValue: string | null
  maxTracks: number
  trackRangeStart: number | null
  trackRangeEnd: number | null
  createdAt: string
  updatedAt: string
}

interface Track {
  id: string
  tracklistId: string
  position: number
  title: string
  artist: string | null
  album: string | null
  genre: string | null
  bpm: number | null
  rating: number | null
  durationSecs: number | null
  musicalKey: string | null
  dateAdded: string | null
  artworkStatus: 'pending' | 'fetched' | 'placeholder' | 'manual'
  artworkUrl: string | null
}

interface ParseWarning {
  row: number
  field: string
  message: string
}
```
