```typescript
import { useEpkStore } from '~/stores/epk'
import type { UpdateEPKContentRequest } from '~/types/epk'

export function useEpkAutosave(delayMs = 1500) {
  const store = useEpkStore()
  let timer: ReturnType<typeof setTimeout> | null = null

  function scheduleSave(patch: UpdateEPKContentRequest): void {
    store.saveStatus = 'saving'
    if (timer) clearTimeout(timer)
    timer = setTimeout(async () => {
      await store.updateContent(patch)
    }, delayMs)
  }

  return { scheduleSave }
}
```

Note: `saveStatus` is NOT returned via storeToRefs() in this composable — it is set directly on the store instance (store.saveStatus = 'saving') because the composable is internal to the store usage. This matches the Phase 1.5 decision: storeToRefs() is for consumer destructuring, not internal store mutation.
