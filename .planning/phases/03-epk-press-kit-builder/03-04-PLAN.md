Create `apps/dj/app/components/epk/EpkPhotosSection.vue`:

```vue
<script setup lang="ts">
import { storeToRefs } from 'pinia'
import { useEpkStore } from '~/stores/epk'

const store = useEpkStore()
const { photoPaths, canAddPhoto, photoCount } = storeToRefs(store)
const fileInput = ref<HTMLInputElement>()

async function onFilesSelected(files: FileList | null) {
  if (!files) return
  for (const file of Array.from(files)) {
    await store.uploadPhoto(file)
  }
}

function onDrop(e: DragEvent) {
  onFilesSelected(e.dataTransfer?.files ?? null)
}
</script>
```
