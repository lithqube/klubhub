export const useGigStore = defineStore('gig', () => {
  // State
  const gigs = ref<Gig[]>([])
  const loading = ref(false)
  const filters = ref({
    status: '',
    venue: '',
    city: '',
    fee_min: null as number | null,
    fee_max: null as number | null,
    from: '',
    to: ''
  })

  // Computed
  const upcomingGigs = computed(() => gigs.value.filter(g => !['played', 'cancelled'].includes(g.status)))
  const ytdEarned = computed(() => /* sum of fee_amount for status=played gigs in current year */)
  const avgFee = computed(() => { /* average fee for played gigs */ })

  // Actions
  async function fetchGigs() {
    loading.value = true
    try {
      const params = new URLSearchParams()
      Object.entries(filters.value).forEach(([k, v]) => { if (v) params.append(k, String(v)) })
      gigs.value = await $fetch(`/api/v1/gigs?${params}`)
    } finally {
      loading.value = false
    }
  }

  async function createGig(gig: GigCreate) { /* POST /api/v1/gigs, unshift to gigs */ }
  async function updateGig(id: string, gig: GigUpdate) { /* PUT /api/v1/gigs/:id */ }
  async function deleteGig(id: string) { /* DELETE /api/v1/gigs/:id, splice from gigs */ }
  async function linkVenue(gigId: string, venueId: string, isPrimary: boolean) { /* POST /api/v1/gigs/:id/link-venue */ }
  async function linkContact(gigId: string, contactId: string, role: string) { /* POST /api/v1/gigs/:id/link-contact */ }
  async function fetchGigsAutocomplete(q: string) { /* GET /api/v1/gigs/autocomplete?q= */ return [] as Gig[] }

  function generateICalUrl() {
    const secret = useRuntimeConfig().public.ICAL_SECRET
    return `/api/v1/gigs/calendar.ics?secret=${secret}`
  }

  return { gigs, loading, filters, upcomingGigs, ytdEarned, avgFee, fetchGigs, createGig, updateGig, deleteGig, linkVenue, linkContact, fetchGigsAutocomplete, generateICalUrl }
})

export const useVenueStore = defineStore('venue', () => {
  const venues = ref<Venue[]>([])
  async function fetchAutocomplete(q: string, limit = 5) {
    return await $fetch(`/api/v1/venues/autocomplete?q=${encodeURIComponent(q)}&limit=${limit}`) as Venue[]
  }
  return { venues, fetchAutocomplete }
})
