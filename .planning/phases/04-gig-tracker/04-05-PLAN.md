## Wave 5: Tracklist ↔ Gig Linking

### Backend — Repository (extend)
Add to `api/internal/gig/repository.go`:

```go
func (r *Repository) LinkTracklist(ctx context.Context, gigID, tracklistID uuid.UUID) error {
    _, err := r.pool.Exec(ctx, `
        INSERT INTO tracklist_gigs (gig_id, tracklist_id)
        VALUES ($1, $2)
        ON CONFLICT (gig_id, tracklist_id) DO NOTHING`,
        gigID, tracklistID,
    )
    return err
}

func (r *Repository) UnlinkTracklist(ctx context.Context, gigID, tracklistID uuid.UUID) error {
    _, err := r.pool.Exec(ctx, `
        DELETE FROM tracklist_gigs WHERE gig_id = $1 AND tracklist_id = $2`,
        gigID, tracklistID,
    )
    return err
}
```

### Backend — Service (extend)
Add to `api/internal/gig/service.go`:

```go
func (s *Service) LinkTracklist(ctx context.Context, gigID, tracklistID uuid.UUID) error {
    // Verify gig exists
    if _, err := s.repo.GetByID(ctx, gigID); err != nil {
        return err
    }
    // Verify tracklist exists (via tracklistRepo)
    if _, _, err := s.tracklistRepo.Get(ctx, tracklistID); err != nil {
        return err
    }
    return s.repo.LinkTracklist(ctx, gigID, tracklistID)
}

func (s *Service) UnlinkTracklist(ctx context.Context, gigID, tracklistID uuid.UUID) error {
    return s.repo.UnlinkTracklist(ctx, gigID, tracklistID)
}
```
